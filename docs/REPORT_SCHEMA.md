# VibeSift Audit Report Schema

This document specifies the normalized data contract for website audit reports generated by VibeSift.

## 1. Schema Definitions

```typescript
export type Severity = 'critical' | 'high' | 'medium' | 'low' | 'info';

export type Category = 
  | 'seo' 
  | 'performance' 
  | 'accessibility' 
  | 'security' 
  | 'cookies' 
  | 'exposure';

export type Confidence = 'certain' | 'high' | 'medium' | 'low';

export type DetectionMethod = 'passive' | 'heuristic' | 'active';

export type EngineType = 'custom-python' | 'lighthouse' | 'owasp-zap' | 'nuclei';

export type VerificationStatus = 'open' | 'fixed' | 'waived' | 'cannot_verify';

export interface FindingEvidence {
  request?: string;
  responseSnippet?: string;
  domSelector?: string;
  observedHeader?: string;
  expectedHeader?: string;
  metrics?: Record<string, string | number>;
  reproductionCurl?: string;
  cwe?: string;
  owaspCategory?: string;
}

export interface FindingRemediation {
  summary: string;
  codeSnippet?: string;
  diffBefore?: string;
  diffAfter?: string;
  frameworkNotes?: string;
  docUrl?: string;
}

export interface Finding {
  id: string;
  scanId: string;
  ruleId: string;
  category: Category;
  severity: Severity;
  confidence: Confidence;
  title: string;
  description: string;
  affectedUrl: string;
  detectionMethod: DetectionMethod;
  engine: EngineType;
  evidence: FindingEvidence;
  impact: string;
  remediation: FindingRemediation;
  verificationStatus: VerificationStatus;
}

export interface ReportSummary {
  totalFindings: number;
  critical: number;
  high: number;
  medium: number;
  low: number;
  info: number;
  readinessScore: number; // 0 to 100
  categoryScores: Record<Category, number>;
}

export interface AuditReport {
  scanId: string;
  ownerUserId: string;
  targetUrl: string;
  normalizedHostname: string;
  scanProfile: 'baseline_public' | 'authorized_deep';
  status: 'QUEUED' | 'VALIDATING' | 'RUNNING' | 'NORMALIZING' | 'COMPLETED' | 'PARTIAL' | 'FAILED' | 'CANCELLED';
  createdAt: string;
  completedAt?: string;
  durationSeconds?: number;
  summary: ReportSummary;
  enginesRun: Array<{
    engine: string;
    displayName: string;
    status: 'passed' | 'partial' | 'failed' | 'skipped_unauthorized' | 'skipped_profile';
    message?: string;
    executionTimeMs?: number;
  }>;
  checksNotPerformed: string[];
  findings: Finding[];
}
```

## 2. Launch Readiness Score Formula

The **Launch Readiness Score** evaluates website technical quality, production readiness, and security posture on a calibrated 0–100 scale:

$$\text{Score} = \max\left(0, \min\left(100, 100 - \sum \text{Penalties}\right)\right)$$

### Severity Weights
- **Critical** (-25 points): Direct security risk, cleartext HTTP on production, exposed `.env` / credentials, invalid TLS certificate.
- **High** (-12 points): Missing or empty `<title>`, active mixed content, outdated TLS protocol, missing image accessibility tags.
- **Medium** (-5 points): Missing HSTS, missing CSP, missing main `<h1>`, broken internal link, missing Cookie `Secure` attribute.
- **Low** (-2 points): Missing `X-Content-Type-Options`, title too long, missing canonical tag, low HSTS max-age.
- **Informational** (0 points): Server header disclosure, technology detection, sitemap tips.

Category scores are computed similarly for each audit domain (`seo`, `performance`, `accessibility`, `security`, `cookies`, `exposure`).
